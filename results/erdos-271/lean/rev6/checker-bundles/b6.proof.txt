  open MeasureTheory Set Real intervalIntegral in (
    let Kker (v : ℝ) : ℝ := Real.exp (-(3 * v / 2)) / (2 - Real.exp (-v)) ^ 3
    let T (τ : ℝ) : Set (ℝ × ℝ) := {p : ℝ × ℝ | 0 ≤ p.2 ∧ p.2 ≤ p.1 ∧ p.1 ≤ τ}




    have T_meas (τ : ℝ) : MeasurableSet (T τ) := by
      show MeasurableSet {p : ℝ × ℝ | 0 ≤ p.2 ∧ p.2 ≤ p.1 ∧ p.1 ≤ τ}
      refine (measurableSet_le measurable_const measurable_snd).inter
        ((measurableSet_le measurable_snd measurable_fst).inter
          (measurableSet_le measurable_fst measurable_const))

    have T_sub (τ : ℝ) : T τ ⊆ Icc 0 τ ×ˢ Icc 0 τ := by
      rintro ⟨t, u⟩ ⟨h1, h2, h3⟩
      exact ⟨⟨le_trans h1 h2, h3⟩, ⟨h1, le_trans h2 h3⟩⟩

    have T_finite (τ : ℝ) : (volume : Measure (ℝ × ℝ)) (T τ) ≠ ⊤ := by
      apply ne_top_of_le_ne_top _ (measure_mono (T_sub τ))
      rw [Measure.volume_eq_prod, Measure.prod_prod]
      exact ENNReal.mul_ne_top (by simp) (by simp)

    have integrableOn_T {f : ℝ × ℝ → ℝ} (hf : Measurable f) (τ M : ℝ)
        (hM : ∀ p ∈ T τ, |f p| ≤ M) : IntegrableOn f (T τ) volume :=
      Measure.integrableOn_of_bounded (T_finite τ) hf.aestronglyMeasurable
        (ae_restrict_of_forall_mem (T_meas τ) (fun p hp => by
          rw [Real.norm_eq_abs]; exact hM p hp))

    have iInt_of_bounded {f : ℝ → ℝ} (hf : Measurable f) {τ : ℝ} (hτ : 0 ≤ τ) (M : ℝ)
        (hM : ∀ x ∈ Icc 0 τ, |f x| ≤ M) : IntervalIntegrable f volume 0 τ := by
      rw [intervalIntegrable_iff_integrableOn_Ioc_of_le hτ]
      exact Measure.integrableOn_of_bounded (by simp) hf.aestronglyMeasurable
        (ae_restrict_of_forall_mem measurableSet_Ioc (fun x hx => by
          rw [Real.norm_eq_abs]; exact hM x (Ioc_subset_Icc_self hx)))

    have abs_le_of_bounds {L U x : ℝ} (h1 : L ≤ x) (h2 : x ≤ U) : |x| ≤ |L| + |U| := by
      rw [abs_le]; constructor
      · have := neg_abs_le L; have := abs_nonneg U; linarith
      · have := le_abs_self U; have := abs_nonneg L; linarith

    have sq_integral_bound {h : ℝ → ℝ} (hm : Measurable h) {L U τ : ℝ} (hτ : 0 ≤ τ)
        (hb : ∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U) :
        ∫ x in (0:ℝ)..τ, h x ^ 2 ≤ U ^ 2 * τ - (L + U) * ∫ x in (0:ℝ)..τ, (U - h x) := by
      have hB : ∀ x ∈ Icc 0 τ, |h x| ≤ |L| + |U| := fun x hx =>
        abs_le_of_bounds (hb x hx).1 (hb x hx).2
      have hi : IntervalIntegrable h volume 0 τ := iInt_of_bounded hm hτ _ hB
      have hi2 : IntervalIntegrable (fun x => h x ^ 2) volume 0 τ :=
        iInt_of_bounded (hm.pow_const 2) hτ ((|L| + |U|) ^ 2) (fun x hx => by
          rw [abs_pow]; exact pow_le_pow_left₀ (abs_nonneg _) (hB x hx) 2)
      have hmono : ∫ x in (0:ℝ)..τ, h x ^ 2 ≤ ∫ x in (0:ℝ)..τ, ((L + U) * h x - L * U) := by
        apply integral_mono_on hτ hi2 ((hi.const_mul _).sub intervalIntegrable_const)
        intro x hx
        obtain ⟨h1, h2⟩ := hb x hx
        nlinarith [mul_nonneg (sub_nonneg.mpr h1) (sub_nonneg.mpr h2)]
      have e1 : ∫ x in (0:ℝ)..τ, ((L + U) * h x - L * U)
          = (L + U) * (∫ x in (0:ℝ)..τ, h x) - L * U * τ := by
        rw [integral_sub (hi.const_mul _) intervalIntegrable_const, intervalIntegral.integral_const_mul,
          intervalIntegral.integral_const]
        simp; ring
      have e2 : ∫ x in (0:ℝ)..τ, (U - h x) = U * τ - ∫ x in (0:ℝ)..τ, h x := by
        rw [integral_sub intervalIntegrable_const hi, intervalIntegral.integral_const]; simp; ring
      rw [e2]; rw [e1] at hmono; nlinarith [hmono]

    have kdiff_bounds {K h : ℝ → ℝ} {L U τ KM : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U))) {p : ℝ × ℝ}
        (hp : p ∈ T τ) :
        0 ≤ K (p.1 - p.2) ∧ K (p.1 - p.2) ≤ KM ∧ p.1 ∈ Icc 0 τ ∧ p.2 ∈ Icc 0 τ := by
      obtain ⟨h1, h2, h3⟩ := hp
      refine ⟨H.2.2.1 _ (by linarith) (by linarith), H.2.2.2.1 _ (by linarith) (by linarith),
        ⟨by linarith, h3⟩, ⟨h1, by linarith⟩⟩

    have Q_lower {K h : ℝ → ℝ} {L U τ KM : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U))) :
        U ^ 2 * (∫ p in T τ, K (p.1 - p.2))
          - U * (∫ p in T τ, K (p.1 - p.2) * (U - h p.1))
          - U * (∫ p in T τ, K (p.1 - p.2) * (U - h p.2))
          ≤ ∫ p in T τ, K (p.1 - p.2) * h p.1 * h p.2 := by
      have hKmp : Measurable (fun p : ℝ × ℝ => K (p.1 - p.2)) :=
        H.1.comp (measurable_fst.sub measurable_snd)
      have hh1 : Measurable (fun p : ℝ × ℝ => h p.1) := H.2.1.comp measurable_fst
      have hh2 : Measurable (fun p : ℝ × ℝ => h p.2) := H.2.1.comp measurable_snd
      set B := |L| + |U| with hBdef
      have hB : ∀ x ∈ Icc 0 τ, |h x| ≤ B := fun x hx =>
        abs_le_of_bounds (H.2.2.2.2 x hx).1 (H.2.2.2.2 x hx).2
      have hKabs : ∀ p ∈ T τ, |K (p.1 - p.2)| ≤ KM := by
        intro p hp; obtain ⟨k0, k1, -, -⟩ := kdiff_bounds H hp
        rw [abs_of_nonneg k0]; exact k1
      have hdB : ∀ x ∈ Icc 0 τ, |U - h x| ≤ |U| + B := fun x hx => by
        have := hB x hx
        calc |U - h x| ≤ |U| + |h x| := abs_sub _ _
          _ ≤ |U| + B := by linarith
      have i0 : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2)) (T τ) volume :=
        integrableOn_T hKmp τ KM hKabs
      have i1 : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.1)) (T τ) volume :=
        integrableOn_T (f := fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.1))
          (hKmp.mul (measurable_const.sub hh1)) τ (KM * (|U| + B)) (fun p hp => by
          obtain ⟨-, -, ht, -⟩ := kdiff_bounds H hp
          show |K (p.1 - p.2) * (U - h p.1)| ≤ _
          rw [abs_mul]
          exact mul_le_mul (hKabs p hp) (hdB _ ht) (abs_nonneg _)
            (le_trans (abs_nonneg _) (hKabs p hp)))
      have i2 : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.2)) (T τ) volume :=
        integrableOn_T (f := fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.2))
          (hKmp.mul (measurable_const.sub hh2)) τ (KM * (|U| + B)) (fun p hp => by
          obtain ⟨-, -, -, hu⟩ := kdiff_bounds H hp
          show |K (p.1 - p.2) * (U - h p.2)| ≤ _
          rw [abs_mul]
          exact mul_le_mul (hKabs p hp) (hdB _ hu) (abs_nonneg _)
            (le_trans (abs_nonneg _) (hKabs p hp)))
      have i3 : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2) * h p.1 * h p.2) (T τ) volume :=
        integrableOn_T (f := fun p : ℝ × ℝ => K (p.1 - p.2) * h p.1 * h p.2)
          ((hKmp.mul hh1).mul hh2) τ (KM * B * B) (fun p hp => by
          obtain ⟨-, -, ht, hu⟩ := kdiff_bounds H hp
          show |K (p.1 - p.2) * h p.1 * h p.2| ≤ _
          rw [abs_mul, abs_mul]
          have hk := hKabs p hp
          have hb1 := hB _ ht
          have hb2 := hB _ hu
          have hB0 : 0 ≤ B := le_trans (abs_nonneg _) hb1
          exact mul_le_mul (mul_le_mul hk hb1 (abs_nonneg _) (le_trans (abs_nonneg _) hk)) hb2
            (abs_nonneg _) (mul_nonneg (le_trans (abs_nonneg _) hk) hB0))
      have hA : IntegrableOn (fun p : ℝ × ℝ => U ^ 2 * K (p.1 - p.2)
          - U * (K (p.1 - p.2) * (U - h p.1))) (T τ) volume :=
        (i0.const_mul _).sub (i1.const_mul _)
      have hA2 : IntegrableOn (fun p : ℝ × ℝ => U ^ 2 * K (p.1 - p.2)
          - U * (K (p.1 - p.2) * (U - h p.1)) - U * (K (p.1 - p.2) * (U - h p.2))) (T τ) volume :=
        hA.sub (i2.const_mul _)
      have hlin : U ^ 2 * (∫ p in T τ, K (p.1 - p.2))
          - U * (∫ p in T τ, K (p.1 - p.2) * (U - h p.1))
          - U * (∫ p in T τ, K (p.1 - p.2) * (U - h p.2))
          = ∫ p in T τ, (U ^ 2 * K (p.1 - p.2) - U * (K (p.1 - p.2) * (U - h p.1))
              - U * (K (p.1 - p.2) * (U - h p.2))) := by
        rw [integral_sub hA (i2.const_mul _), integral_sub (i0.const_mul _) (i1.const_mul _),
          MeasureTheory.integral_const_mul, MeasureTheory.integral_const_mul,
          MeasureTheory.integral_const_mul]
      rw [hlin]
      apply setIntegral_mono_on hA2 i3 (T_meas τ)
      intro p hp
      obtain ⟨k0, -, ht, hu⟩ := kdiff_bounds H hp
      have d1 : 0 ≤ U - h p.1 := sub_nonneg.mpr (H.2.2.2.2 _ ht).2
      have d2 : 0 ≤ U - h p.2 := sub_nonneg.mpr (H.2.2.2.2 _ hu).2
      have key : K (p.1 - p.2) * ((U - h p.1) * (U - h p.2)) ≥ 0 :=
        mul_nonneg k0 (mul_nonneg d1 d2)
      nlinarith [key]

    have lemma4_of_kernel_bounds {K h : ℝ → ℝ} {L U τ KM β η : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U)))
        (hτ : 0 ≤ τ) (hU : 0 ≤ U)
        (hQ1 : β * τ - η ≤ ∫ p in T τ, K (p.1 - p.2))
        (hC1 : ∫ p in T τ, K (p.1 - p.2) * (U - h p.1) ≤ β * ∫ x in (0:ℝ)..τ, (U - h x))
        (hC2 : ∫ p in T τ, K (p.1 - p.2) * (U - h p.2) ≤ β * ∫ x in (0:ℝ)..τ, (U - h x)) :
        (3 / 2) * (∫ x in (0:ℝ)..τ, h x ^ 2) - 4 * (∫ p in T τ, K (p.1 - p.2) * h p.1 * h p.2)
          ≤ (3 / 2 - 4 * β) * U ^ 2 * τ
            - (2 * (3 / 2 - 4 * β) * U - (3 / 2) * (U - L)) * (∫ x in (0:ℝ)..τ, (U - h x))
            + 4 * U ^ 2 * η := by
      have ha := sq_integral_bound H.2.1 hτ H.2.2.2.2
      have hq := Q_lower H
      set D := ∫ x in (0:ℝ)..τ, (U - h x)
      set Q1 := ∫ p in T τ, K (p.1 - p.2)
      set C1 := ∫ p in T τ, K (p.1 - p.2) * (U - h p.1)
      set C2 := ∫ p in T τ, K (p.1 - p.2) * (U - h p.2)
      have hU2 : 0 ≤ U ^ 2 := sq_nonneg U
      have s1 : U ^ 2 * (β * τ - η) ≤ U ^ 2 * Q1 := mul_le_mul_of_nonneg_left hQ1 hU2
      have s2 : U * C1 ≤ U * (β * D) := mul_le_mul_of_nonneg_left hC1 hU
      have s3 : U * C2 ≤ U * (β * D) := mul_le_mul_of_nonneg_left hC2 hU
      nlinarith [ha, hq, s1, s2, s3]






    have tri_iter {τ : ℝ} {F : ℝ × ℝ → ℝ} (hF : IntegrableOn F (T τ) volume) :
        ∫ p in T τ, F p = ∫ t in Icc 0 τ, ∫ u in Icc 0 t, F (t, u) := by
      rw [← MeasureTheory.integral_indicator (T_meas τ)]
      have hI : Integrable ((T τ).indicator F) (volume.prod volume) := by
        rw [← Measure.volume_eq_prod]; exact hF.integrable_indicator (T_meas τ)
      rw [Measure.volume_eq_prod, integral_prod _ hI]
      rw [← MeasureTheory.integral_indicator measurableSet_Icc]
      congr 1
      ext t
      by_cases ht : t ≤ τ
      · by_cases h0 : 0 ≤ t
        · rw [indicator_of_mem (show t ∈ Icc 0 τ from ⟨h0, ht⟩)]
          rw [← MeasureTheory.integral_indicator measurableSet_Icc]
          congr 1; ext u
          simp only [indicator, T, mem_ofPred_eq, mem_Icc]
          by_cases hu : 0 ≤ u ∧ u ≤ t
          · simp [hu.1, hu.2, ht]
          · have : ¬ (0 ≤ u ∧ u ≤ t ∧ t ≤ τ) := fun h => hu ⟨h.1, h.2.1⟩
            simp only [this, hu, ite_false]
        · rw [indicator_of_notMem (show t ∉ Icc 0 τ from fun h => h0 h.1)]
          have : ∀ u, (T τ).indicator F (t, u) = 0 := by
            intro u
            apply indicator_of_notMem
            rintro ⟨h1, h2, -⟩
            exact h0 (le_trans h1 h2)
          simp [this]
      · rw [indicator_of_notMem (show t ∉ Icc 0 τ from fun h => ht h.2)]
        have : ∀ u, (T τ).indicator F (t, u) = 0 := by
          intro u
          apply indicator_of_notMem
          rintro ⟨-, -, h3⟩
          exact ht h3
        simp [this]

    have tri_iter_swap {τ : ℝ} {F : ℝ × ℝ → ℝ} (hF : IntegrableOn F (T τ) volume) :
        ∫ p in T τ, F p = ∫ u in Icc 0 τ, ∫ t in Icc u τ, F (t, u) := by
      rw [← MeasureTheory.integral_indicator (T_meas τ)]
      have hI : Integrable ((T τ).indicator F) (volume.prod volume) := by
        rw [← Measure.volume_eq_prod]; exact hF.integrable_indicator (T_meas τ)
      rw [Measure.volume_eq_prod, integral_prod_symm _ hI]
      rw [← MeasureTheory.integral_indicator measurableSet_Icc]
      congr 1
      ext u
      by_cases hu : 0 ≤ u ∧ u ≤ τ
      · rw [indicator_of_mem (show u ∈ Icc 0 τ from hu)]
        rw [← MeasureTheory.integral_indicator measurableSet_Icc]
        congr 1; ext t
        simp only [indicator, T, mem_ofPred_eq, mem_Icc]
        by_cases ht : u ≤ t ∧ t ≤ τ
        · simp [ht.1, ht.2, hu.1]
        · simp only [ht, and_false, ↓reduceIte]
      · rw [indicator_of_notMem (show u ∉ Icc 0 τ from hu)]
        have : ∀ t, (T τ).indicator F (t, u) = 0 := by
          intro t
          apply indicator_of_notMem
          rintro ⟨h1, h2, h3⟩
          exact hu ⟨h1, le_trans h2 h3⟩
        simp [this]

    have iInt_eq_Icc {f : ℝ → ℝ} {τ : ℝ} (hτ : 0 ≤ τ) :
        ∫ x in (0:ℝ)..τ, f x = ∫ x in Icc 0 τ, f x := by
      rw [intervalIntegral.integral_of_le hτ, integral_Icc_eq_integral_Ioc]

    have outer_bound {g m : ℝ → ℝ} {τ β : ℝ} (hτ : 0 ≤ τ) (hg : Measurable g) (G : ℝ)
        (hgb : ∀ x ∈ Icc 0 τ, 0 ≤ g x ∧ g x ≤ G) (hm0 : ∀ x ∈ Icc 0 τ, 0 ≤ m x)
        (hmβ : ∀ x ∈ Icc 0 τ, m x ≤ β) :
        ∫ x in Icc 0 τ, g x * m x ≤ β * ∫ x in (0:ℝ)..τ, g x := by
      rw [iInt_eq_Icc hτ, ← MeasureTheory.integral_const_mul]
      have hgi : IntegrableOn (fun x => β * g x) (Icc 0 τ) volume := by
        refine Measure.integrableOn_of_bounded (M := |β| * G) (by simp)
          (hg.const_mul β).aestronglyMeasurable
          (ae_restrict_of_forall_mem measurableSet_Icc (fun x hx => ?_))
        rw [Real.norm_eq_abs, abs_mul, abs_of_nonneg (hgb x hx).1]
        exact mul_le_mul_of_nonneg_left (hgb x hx).2 (abs_nonneg _)
      apply integral_mono_of_nonneg _ hgi
      · exact ae_restrict_of_forall_mem measurableSet_Icc (fun x hx => by
          have := hgb x hx
          show g x * m x ≤ β * g x
          nlinarith [hmβ x hx, this.1])
      · exact ae_restrict_of_forall_mem measurableSet_Icc (fun x hx =>
          mul_nonneg (hgb x hx).1 (hm0 x hx))

    have C1_le {K h : ℝ → ℝ} {L U τ KM β : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U))) (hτ : 0 ≤ τ)
        (hβ : ∀ t ∈ Icc 0 τ, ∫ u in Icc 0 t, K (t - u) ≤ β) :
        ∫ p in T τ, K (p.1 - p.2) * (U - h p.1) ≤ β * ∫ x in (0:ℝ)..τ, (U - h x) := by
      have hint : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.1)) (T τ) volume := by
        have hKmp : Measurable (fun p : ℝ × ℝ => K (p.1 - p.2)) :=
          H.1.comp (measurable_fst.sub measurable_snd)
        refine integrableOn_T (f := fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.1))
          (hKmp.mul (measurable_const.sub (H.2.1.comp measurable_fst))) τ (KM * (U - L))
          (fun p hp => ?_)
        obtain ⟨k0, k1, ht, -⟩ := kdiff_bounds H hp
        have := H.2.2.2.2 _ ht
        show |K (p.1 - p.2) * (U - h p.1)| ≤ _
        rw [abs_mul, abs_of_nonneg k0, abs_of_nonneg (by linarith)]
        exact mul_le_mul k1 (by linarith) (by linarith) (le_trans k0 k1)
      rw [tri_iter hint]
      have hinner : ∀ t, ∫ u in Icc 0 t, K (t - u) * (U - h t)
          = (U - h t) * ∫ u in Icc 0 t, K (t - u) := by
        intro t; rw [MeasureTheory.integral_mul_const, mul_comm]
      simp_rw [hinner]
      refine outer_bound hτ (measurable_const.sub H.2.1) (U - L)
        (fun x hx => ⟨by linarith [(H.2.2.2.2 x hx).2], by linarith [(H.2.2.2.2 x hx).1]⟩) ?_ hβ
      intro t ht
      apply setIntegral_nonneg measurableSet_Icc
      intro u hu
      exact H.2.2.1 _ (by linarith [hu.2]) (by linarith [hu.1, ht.2])

    have C2_le {K h : ℝ → ℝ} {L U τ KM β : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U))) (hτ : 0 ≤ τ)
        (hβ : ∀ u ∈ Icc 0 τ, ∫ t in Icc u τ, K (t - u) ≤ β) :
        ∫ p in T τ, K (p.1 - p.2) * (U - h p.2) ≤ β * ∫ x in (0:ℝ)..τ, (U - h x) := by
      have hint : IntegrableOn (fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.2)) (T τ) volume := by
        have hKmp : Measurable (fun p : ℝ × ℝ => K (p.1 - p.2)) :=
          H.1.comp (measurable_fst.sub measurable_snd)
        refine integrableOn_T (f := fun p : ℝ × ℝ => K (p.1 - p.2) * (U - h p.2))
          (hKmp.mul (measurable_const.sub (H.2.1.comp measurable_snd))) τ (KM * (U - L))
          (fun p hp => ?_)
        obtain ⟨k0, k1, -, hu⟩ := kdiff_bounds H hp
        have := H.2.2.2.2 _ hu
        show |K (p.1 - p.2) * (U - h p.2)| ≤ _
        rw [abs_mul, abs_of_nonneg k0, abs_of_nonneg (by linarith)]
        exact mul_le_mul k1 (by linarith) (by linarith) (le_trans k0 k1)
      rw [tri_iter_swap hint]
      have hinner : ∀ u, ∫ t in Icc u τ, K (t - u) * (U - h u)
          = (U - h u) * ∫ t in Icc u τ, K (t - u) := by
        intro u; rw [MeasureTheory.integral_mul_const, mul_comm]
      simp_rw [hinner]
      refine outer_bound hτ (measurable_const.sub H.2.1) (U - L)
        (fun x hx => ⟨by linarith [(H.2.2.2.2 x hx).2], by linarith [(H.2.2.2.2 x hx).1]⟩) ?_ hβ
      intro u hu
      apply setIntegral_nonneg measurableSet_Icc
      intro t ht
      exact H.2.2.1 _ (by linarith [ht.1]) (by linarith [ht.2, hu.1])

    have lemma4_reduced {K h : ℝ → ℝ} {L U τ KM β η : ℝ} (H : (Measurable K ∧ Measurable h ∧ (∀ v, 0 ≤ v → v ≤ τ → 0 ≤ K v) ∧ (∀ v, 0 ≤ v → v ≤ τ → K v ≤ KM) ∧ (∀ x ∈ Icc 0 τ, L ≤ h x ∧ h x ≤ U)))
        (hτ : 0 ≤ τ) (hU : 0 ≤ U)
        (hQ1 : β * τ - η ≤ ∫ p in T τ, K (p.1 - p.2))

    ∀ (K M : Real → Real), (∀ v, K v = Real.exp (-3 * v / 2) / (2 - Real.exp (-v)) ^ 3) → (∀ v, M v = v * K v) → MeasureTheory.IntegrableOn M (Set.Ioi 0) MeasureTheory.volume ∧ 0 ≤ MeasureTheory.integral (MeasureTheory.Measure.restrict MeasureTheory.volume (Set.Ioi 0)) M ∧ MeasureTheory.integral (MeasureTheory.Measure.restrict MeasureTheory.volume (Set.Ioi 0)) M ≤ 2 :=
    by
      intro K M hK hM
      have hmeas : Measurable M := by
        have heq : M = fun v : Real => v * (Real.exp (-3 * v / 2) / (2 - Real.exp (-v)) ^ 3) := by
          funext v
          rw [hM v, hK v]
        rw [heq]
        fun_prop
      have hbound : ∀ v : Real, 0 ≤ v → 0 ≤ M v ∧ M v ≤ 2 * Real.exp (-v) := by
        intro v hv
        have he : Real.exp (-v) ≤ 1 := Real.exp_le_one_iff.mpr (neg_nonpos.mpr hv)
        have hd : 1 ≤ 2 - Real.exp (-v) := by linarith
        have hd0 : 0 ≤ 2 - Real.exp (-v) := by linarith
        have hd2 : 1 ≤ (2 - Real.exp (-v)) ^ 2 := by
          nlinarith [sq_nonneg (2 - Real.exp (-v) - 1)]
        have hd3 : 1 ≤ (2 - Real.exp (-v)) ^ 3 := by
          nlinarith [mul_nonneg (sub_nonneg.mpr hd) (sq_nonneg (2 - Real.exp (-v)))]
        have hk0 : 0 ≤ K v := by
          rw [hK v]
          exact div_nonneg (Real.exp_nonneg _) (pow_nonneg hd0 _)
        have hk1 : K v ≤ Real.exp (-3 * v / 2) := by
          rw [hK v]
          exact div_le_self (Real.exp_nonneg _) hd3
        have hvexp : v ≤ 2 * Real.exp (v / 2) := by
          have h := Real.add_one_le_exp (v / 2)
          linarith
        have hex : Real.exp (v / 2) * Real.exp (-3 * v / 2) = Real.exp (-v) := by
          rw [← Real.exp_add]
          congr 1
          ring
        constructor
        · rw [hM v]
          exact mul_nonneg hv hk0
        · calc
            M v = v * K v := hM v
            _ ≤ v * Real.exp (-3 * v / 2) := mul_le_mul_of_nonneg_left hk1 hv
            _ ≤ (2 * Real.exp (v / 2)) * Real.exp (-3 * v / 2) := mul_le_mul_of_nonneg_right hvexp (Real.exp_nonneg _)
            _ = 2 * Real.exp (-v) := by rw [mul_assoc, hex]
      have hg : MeasureTheory.IntegrableOn (fun v : Real => 2 * Real.exp (-v)) (Set.Ioi 0) MeasureTheory.volume :=
        (integrableOn_exp_neg_Ioi 0).const_mul 2
      have hmi : MeasureTheory.IntegrableOn M (Set.Ioi 0) MeasureTheory.volume := by
        apply hg.mono' hmeas.aestronglyMeasurable
        filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioi] with v hv
        have hb := hbound v (le_of_lt hv)
        rw [Real.norm_eq_abs, abs_of_nonneg hb.1]
        exact hb.2
      refine ⟨hmi, ?_, ?_⟩
      · apply MeasureTheory.integral_nonneg_of_ae
        filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioi] with v hv
        exact (hbound v (le_of_lt hv)).1
      · have hcmp := MeasureTheory.integral_mono_ae hmi hg (by
          filter_upwards [MeasureTheory.ae_restrict_mem measurableSet_Ioi] with v hv
          exact (hbound v (le_of_lt hv)).2)
        have hgi : MeasureTheory.integral (MeasureTheory.Measure.restrict MeasureTheory.volume (Set.Ioi 0)) (fun v : Real => 2 * Real.exp (-v)) = 2 := by
          rw [MeasureTheory.integral_const_mul, integral_exp_neg_Ioi_zero]
          ring
        rw [hgi] at hcmp
        exact hcmp

    have Kker_spec (v : ℝ) : Kker v = Real.exp (-3 * v / 2) / (2 - Real.exp (-v)) ^ 3 := by
      unfold Kker; congr 2; ring


    have lemma4_concrete {h : ℝ → ℝ} {L U t0 T : ℝ} (hm : Measurable h) (hT : t0 ≤ T)
        (hU : 0 ≤ U) (hb : ∀ x ∈ Icc t0 T, L ≤ h x ∧ h x ≤ U) :
        let β : ℝ := 3 / 8 - Real.log (1 + Real.sqrt 2) / (8 * Real.sqrt 2)
        let η : ℝ := ∫ v in Ioi 0, v * Kker v
        0 ≤ η ∧ η ≤ 2 ∧
        (3 / 2) * (∫ x in t0..T, h x ^ 2)
          - 4 * (∫ t in Ioo t0 T, h t * ∫ u in Ioo t0 t, Kker (t - u) * h u)
          ≤ (3 / 2 - 4 * β) * U ^ 2 * (T - t0)
            - (2 * (3 / 2 - 4 * β) * U - (3 / 2) * (U - L)) * (∫ x in t0..T, (U - h x))
            + 4 * U ^ 2 * η := by
      intro β η
      obtain ⟨hKi, hβ⟩ := kernel_formula Kker Kker_spec
      obtain ⟨hMi, h0, h2⟩ := FirstMoment_checked Kker (fun v => v * Kker v)
        Kker_spec (fun v => rfl)
      exact ⟨h0, h2, lemma4_window hm hT hU hb hKi hβ hMi rfl⟩

    intro K h L U t0 T' sq d mom Kt G hK hsq hd hmom hKt hG hm hT hU hb
    obtain rfl : sq = fun x => h x ^ 2 := funext hsq
    obtain rfl : d = fun x => U - h x := funext hd
    have hKK : K = Kker := funext (fun v => by rw [hK v])
    subst hKK
    obtain rfl : mom = fun v => v * Kker v := funext hmom
    obtain rfl : Kt = fun t u => Kker (t - u) * h u := funext (fun t => funext (fun u => hKt t u))
    obtain rfl : G = fun t => h t * MeasureTheory.integral (MeasureTheory.Measure.restrict MeasureTheory.volume (Set.Ioo t0 t)) (fun u => Kker (t - u) * h u) := funext hG
    exact lemma4_concrete hm hT hU hb)
